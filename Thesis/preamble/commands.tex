% Insert a figure that obeys the format rules (centered, width <= text width,
% caption centered below).  Usage:
%   \fig{filename}{width-fraction}{Caption text.}{fig:label}
\newcommand{\fig}[4]{%
  \begin{figure}[htbp]
    \centering
    \includegraphics[width=#2\textwidth]{#1}
    \caption{#3}
    \label{#4}
  \end{figure}%
}

% Code excerpts.  The template recognises only two kinds of numbered element -
% Figures and Tables - so a code excerpt is placed inside a *figure* float and
% captioned "Figure x.y: ..." (centered, below), exactly like any other figure,
% rather than with a separate "Listing" counter.  The macro below only trims the
% space the listing frame adds inside a float.
\newcommand{\codegap}{\vspace{-0.6em}}
